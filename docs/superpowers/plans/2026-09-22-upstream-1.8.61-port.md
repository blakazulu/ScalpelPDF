```csharp
using System;

namespace Scalpel.Services
{
    /// <summary>
    /// The canvas coordinate space annotations are stored in: the page's visual size scaled so its
    /// longest side is 2048. Continuous view always uses it; single-page rendering matches it. Save
    /// falls back to it for a page that was never rendered this session, instead of skipping the
    /// page and silently dropping its annotations.
    /// </summary>
    public static class CanonicalRenderDims
    {
        public const int LongSide = 2048;

        public static (int w, int h) For(double widthPt, double heightPt, int rotation)
        {
            int rot = ((rotation % 360) + 360) % 360;
            double pw = Math.Max(1, widthPt), ph = Math.Max(1, heightPt);
            if (rot == 90 || rot == 270) (pw, ph) = (ph, pw);
            double max = Math.Max(pw, ph);
            return (Math.Max(1, (int)Math.Round(LongSide * pw / max)),
                    Math.Max(1, (int)Math.Round(LongSide * ph / max)));
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;

namespace Scalpel.Services
{
    /// <summary>
    /// Moves per-page annotation lists to their new page numbers after a page operation, so
    /// unsaved annotations follow their page instead of being thrown away on reload.
    /// </summary>
    public static class AnnotationPageRemap
    {
        public static void Apply<T>(Dictionary<int, List<T>> byPage, Func<int, int?> map, Action<T, int> setPage)
        {
            var moved = new Dictionary<int, List<T>>();
            foreach (var kv in byPage)
            {
                int? target = map(kv.Key);
                if (target is null) continue;                       // page deleted
                if (!moved.TryGetValue(target.Value, out var list)) moved[target.Value] = list = [];
                foreach (var item in kv.Value) { setPage(item, target.Value); list.Add(item); }
            }
            byPage.Clear();
            foreach (var kv in moved) byPage[kv.Key] = kv.Value;
        }

        public static Func<int, int?> Insert(int at, int count) =>
            p => p >= at ? p + count : p;

        public static Func<int, int?> Delete(IEnumerable<int> removed)
        {
            var gone = new HashSet<int>(removed);
            var sorted = gone.OrderBy(x => x).ToArray();
            return p => gone.Contains(p) ? null : p - sorted.Count(r => r < p);
        }

        /// <summary>The page at <paramref name="from"/> ends up at <paramref name="to"/>
        /// (remove then insert, the way PdfPages.RemoveAt + Insert behaves).</summary>
        public static Func<int, int?> Move(int from, int to) => p =>
        {
            if (p == from) return to;
            if (from < to && p > from && p <= to) return p - 1;
            if (from > to && p >= to && p < from) return p + 1;
            return p;
        };
    }
}
```

```csharp
public static void Save(PdfDocument doc, string path)
{
    PrepareForSave(doc);
    string full = Path.GetFullPath(path);
    string dir = Path.GetDirectoryName(full) ?? ".";
    string temp = Path.Combine(dir, "." + Path.GetFileName(full) + "." + Guid.NewGuid().ToString("N").Substring(0, 8) + ".tmp");

        {

            catch (IOException) { File.Copy(temp, full, overwrite: true); }
            catch (UnauthorizedAccessException) { File.Copy(temp, full, overwrite: true); }
        }
        else File.Move(temp, full);
    }
    finally
    {
        try { if (File.Exists(temp)) File.Delete(temp); } catch { }
    }
}
```

```csharp
    {
```

```csharp
        try { if (o.DestinationPage is not null) idx = IndexOfPage(src, o.DestinationPage); } catch { }
        if (idx < 0 || idx >= dst.PageCount)
        {
            // Dead destination: keep its children one level up (upstream sanitiser rule).
            if (o.HasChildren) CopyOutlines(src, o.Outlines, dst, to, depth + 1);
            continue;
        }
        var copy = to.Add(o.Title, dst.Pages[idx], o.Opened);
        if (o.HasChildren) CopyOutlines(src, o.Outlines, dst, copy.Outlines, depth + 1);
    }
}

private static int IndexOfPage(PdfDocument doc, PdfPage page)
{
    for (int i = 0; i < doc.PageCount; i++) if (ReferenceEquals(doc.Pages[i], page)) return i;
    return -1;
}
```

```csharp
public static void DrawFullPage(PdfDocument doc, PdfPage page, byte[] bgra, int w, int h, bool oneBit)
{
    byte[] raw = oneBit ? PackBits(bgra, w, h) : ToGray8(bgra, w, h);
    var img = new PdfDictionary(doc);
    img.Elements["/Type"] = new PdfName("/XObject");
    img.Elements["/Subtype"] = new PdfName("/Image");
    img.Elements["/Width"] = new PdfInteger(w);
    img.Elements["/Height"] = new PdfInteger(h);
    img.Elements["/ColorSpace"] = new PdfName("/DeviceGray");
    img.Elements["/BitsPerComponent"] = new PdfInteger(oneBit ? 1 : 8);
    img.Elements["/Filter"] = new PdfName("/FlateDecode");
    img.CreateStream(Zlib(raw));
    doc.Internals.AddObject(img);

    var xobjs = page.Resources.Elements.GetDictionary("/XObject");
    if (xobjs is null) { xobjs = new PdfDictionary(doc); page.Resources.Elements["/XObject"] = xobjs; }
    xobjs.Elements["/ScFlat0"] = img.Reference;

    string pw = page.Width.Point.ToString("0.###", CultureInfo.InvariantCulture);
    string ph = page.Height.Point.ToString("0.###", CultureInfo.InvariantCulture);
    var content = page.Contents.AppendContent();
    content.CreateStream(Encoding.ASCII.GetBytes($"q {pw} 0 0 {ph} 0 0 cm /ScFlat0 Do Q"));
}
```
